def verify_answer(question, answer, retrieved_docs, all_thoughts):
    """
    验证答案是否满足所有约束条件
    """
    # 1. 提取问题中的所有约束
    constraints = extract_all_constraints(question)
    
    # 2. 检查每个约束是否在检索文档或推理过程中被满足
    satisfied = []
    missing = []
    
    for constraint in constraints:
        if check_constraint_in_docs(constraint, retrieved_docs, all_thoughts):
            satisfied.append(constraint)
        else:
            missing.append(constraint)
    
    # 3. 如果关键约束缺失，返回"信息不足"
    if len(missing) > len(constraints) * 0.3:  # 超过30%约束缺失
        return None, f"Missing information for: {missing}"
    
    return answer, "Verified"
```

**预期效果**: 减少错误答案，提高系统可靠性

def clean_answer(answer):
    """
    清理答案，提取关键信息
    """
    # 移除常见前缀
    prefixes = ["So the answer is:", "The answer is:", "Answer:"]
    for prefix in prefixes:
        if answer.startswith(prefix):
            answer = answer[len(prefix):].strip()
    
    # 只保留第一句或前50个字符
    if len(answer) > 100:
        first_period = answer.find('.')
        if first_period > 0:
            answer = answer[:first_period].strip()
        else:
            answer = answer[:50].strip()
    
    return answer
```
